
Создание и заполнение таблицы difficult:
CREATE TABLE IF NOT EXISTS basic.difficult(
    id integer NOT NULL,
    level character varying(100) COLLATE pg_catalog."default",
    CONSTRAINT difficult_pkey PRIMARY KEY (id)
);
TABLESPACE pg_default;
INSERT INTO basic.difficult(id, level) VALUES (1, 'easy'), (2, 'norm'),(3, 'hard');
select * from basic."difficult";


Создание и заполнение таблицы questions:
CREATE TABLE IF NOT EXISTS basic.questions(
    id integer NOT NULL,
    question character varying(100) COLLATE pg_catalog."default",
    difficult_id integer,
    CONSTRAINT questions_pkey PRIMARY KEY (id),
    CONSTRAINT difficult_id FOREIGN KEY (difficult_id)
        REFERENCES basic.difficult (id) MATCH SIMPLE
        ON UPDATE NO ACTION
        ON DELETE NO ACTION
        NOT VALID
)
TABLESPACE pg_default;

Создание и заполнение таблицы answers:
CREATE TABLE IF NOT EXISTS basic.answers
(
    id integer NOT NULL,
    answer_option character varying(100) COLLATE pg_catalog."default",
    question_id integer NOT NULL,
    correct boolean,
    CONSTRAINT question_id FOREIGN KEY (question_id)
        REFERENCES basic.questions (id) MATCH SIMPLE
        ON UPDATE NO ACTION
        ON DELETE NO ACTION
        NOT VALID
)
TABLESPACE pg_default;
